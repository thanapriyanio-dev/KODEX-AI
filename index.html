    public static void solvePattern(String str) {
        int len = str.length();
        for (int i = 0; i < len; i++) {
            for (int j = 0; j < len; j++) {
                if (j == i || j == (len - 1 - i)) {
                    System.out.print(str.charAt(j));
                } else {
                    System.out.print(" ");
                }
            }
            System.out.println();
        }
    }
